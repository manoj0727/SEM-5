% Simple PSK Constellation - Save as psk_analysis.m

clc; clear; close all;

N = 1000; SNR = 20;

% Modulation
sig1 = 2*randi([0 1],N,1) - 1;                    % BPSK
sig2 = exp(1j*pi/4*(2*randi([0 3],N,1) + 1));    % QPSK
sig3 = sig2;                                      % OQPSK

% Add noise
rx1 = awgn(sig1, SNR);
rx2 = awgn(sig2, SNR);
rx3 = awgn(sig3, SNR);

% Plot
subplot(1,3,1);
plot(real(rx1), imag(rx1), 'b.');
title('BPSK'); grid on; axis([-2 2 -2 2]);

subplot(1,3,2);
plot(real(rx2), imag(rx2), 'r.');
title('QPSK'); grid on; axis([-2 2 -2 2]);

subplot(1,3,3);
plot(real(rx3), imag(rx3), 'g.');
title('OQPSK'); grid on; axis([-2 2 -2 2]);
