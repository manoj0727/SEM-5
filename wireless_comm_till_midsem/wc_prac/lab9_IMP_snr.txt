% Practical 9: Antenna Array → Array Gain → Better SNR (LOS)

clc; clear; close all;

N  = [1 2 4 8 16];     % antennas
Pt = 1;                % Tx power
N0 = 0.1;              % noise power

SNR1 = Pt/N0;          % SNR for 1 antenna
SNRn = N*SNR1;         % array gain rule: ≈N×SNR1
SNRdB = 10*log10(SNRn);


figure;
plot(N, SNRdB, 'Marker','o','LineWidth',1.5);
grid on;
xlabel('Number of antennas');
ylabel('SNR (dB)');
title('SNR ↑ by Array Gain (LOS)');
