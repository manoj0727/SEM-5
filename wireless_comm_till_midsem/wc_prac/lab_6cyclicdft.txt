% --- PARAMETERS ---
N  = 8;                % DFT/FFT size
cp = 2;                % cyclic prefix length (≥ len(h)-1)
s  = randi([0 10],1,N); % Tx signal
h  = randi([0 5],1,3);  % channel impulse response (len = 3)

% --- CONVOLUTIONS ---
linConv = conv(s,h);           % Linear convolution
circDFT = ifft(fft(s,N).*fft(h,N)); % IDFT(DFT(s)*DFT(h)) → circular conv
rxCP   = conv([s(end-cp+1:end) s], h); % channel output (after adding CP)
rxN    = rxCP(cp+1:cp+N);       % remove CP → take N samples

% --- DISPLAY ---
disp('--- Values ---');
disp([s' circDFT.' real(rxN).']); % show few samples quickly

% --- ERRORS ---
e1 = norm(circDFT - rxN);  % error between theory and practical result
fprintf('Error (theory vs CP-rx) = %.2e\n', e1);

% --- PLOT (Single figure) ---
figure('Name','Cyclic Prefix Test','NumberTitle','off');
stem(real(rxN),'LineWidth',1.2);
grid on;
title('Received Signal After Removing CP (Should Match Circular Conv)');
xlabel('n'); ylabel('Amplitude');
