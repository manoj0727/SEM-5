%% Practical 7: Log-Distance Path Loss + Log-Normal Shadowing (Wi-Fi LOS Obstacle)
% Manoj Kumawat (UI23EC36)

clc; clear; close all;
fprintf('Log-Distance Model + Shadowing – Manoj Kumawat\n');

% --- Parameters ---
f  = 2.4e9;                % freq (Hz)
d0 = 1;                    % ref distance (m)
d  = 100*(1:0.2:100);       % test distances (m)
Pt = 0;                    % Tx power (dBm)
Gt = 1; Gr = 1;             % antenna gains (dBi)
c  = 3e8;                  % speed of light
sigma = 2;                 % shadowing SD (dB)
n = 3.5;                   % path-loss exponent
L = 1;                     % system loss (no extra loss)

% --- Calculations ---
lambda = c/f;
PL0 = 20*log10(4*pi*d0/lambda);   % reference path loss
X  = sigma*randn(size(d));        % random shadowing
PL = PL0 + 10*n*log10(d/d0) + X;  % total PL (dB)
Pr = Pt + Gt + Gr - PL;           % received power (dBm)

% Show first 5 rows (simple matrix for easy memory)
disp([d(1:5)' PL(1:5)' Pr(1:5)']);

% --- Plot ---
figure;
plot(log10(d), Pr, 'LineWidth',1.5);
grid on;
xlabel('log10(d) [m]');
ylabel('Pr [dBm]');
title('Path Loss + Shadowing (LOS)');
