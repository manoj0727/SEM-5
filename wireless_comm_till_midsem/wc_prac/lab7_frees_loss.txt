clc; clear; close all;
fprintf('Wi-Fi Friis Model – Manoj Kumawat (UI23EC36)\n');

f  = 2.4e9;                 % 2.4 GHz band
d  = 1:1:100;               % distances (m)
Pt = 0;                     % Tx power (dBm)
Gt = 1; Gr = 1;              % antenna gains (dBi)
c  = 3e8;                   % speed of light

lambda = c/f;               % wavelength
FSPL = 20*log10(4*pi*d/lambda); % path loss (dB)

Pr = Pt + Gt + Gr - FSPL;   % received power (dBm)

% Show first 5 values
disp([d(1:5)' FSPL(1:5)' Pr(1:5)']);

% Plot
figure;
plot(d, Pr, 'LineWidth', 1.5);
grid on;
xlabel('Distance (m)'); ylabel('Pr (dBm)');
title('SNR ↑ by Array Gain in LOS Wi-Fi Propagation');
