%% Task 2: OFDM with Frequency Domain Equalization

numSubcarriers = 64;    % OFDM subcarriers
numOFDM = 1000;         % Number of OFDM symbols
modOrder = 4;           % QPSK

% Transmitter bits
data = randi([0 modOrder-1], numOFDM, numSubcarriers);

txSymbols = pskmod(data, modOrder, pi/modOrder);     % QPSK symbols

% OFDM modulation (IFFT)
txOFDM = ifft(txSymbols, [], 2);

% Multipath channel
channel = [0.8 0.3 0.1];
H = fft(channel, numSubcarriers);   % Channel frequency response

% Pass through channel + noise
rxSignal = zeros(size(txOFDM));
for k = 1:numOFDM
    convSig = conv(txOFDM(k,:), channel, 'same');
    noise = 0.01*(randn(1,numSubcarriers) + 1j*randn(1,numSubcarriers));
    rxSignal(k,:) = convSig + noise;
end

% OFDM demodulation
rxFFT = fft(rxSignal, [], 2);

% Equalization
equalized = rxFFT ./ H;

% QPSK demodulation
rxData = pskdemod(equalized, modOrder, pi/modOrder);

% BER
numErrors = sum(data(:) ~= rxData(:));
BER = numErrors / (numOFDM*numSubcarriers);

fprintf('OFDM BER with frequency domain equalization = %e\n', BER);
