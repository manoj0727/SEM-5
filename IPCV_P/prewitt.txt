import cv2
import numpy as np
from google.colab.patches import cv2_imshow

# Step 1: Read color image
img = cv2.imread('/content/sample_data/Lena.jpg')

# Step 2: Convert RGB to Gray
gray = cv2.cvtColor(img, cv2.COLOR_BGR2GRAY)

# Step 3: Define Prewitt Masks
prewitt_x = np.array([[-1, 0, 1],
                      [-1, 0, 1],
                      [-1, 0, 1]], dtype=np.float32)

prewitt_y = np.array([[-1, -1, -1],
                      [ 0,  0,  0],
                      [ 1,  1,  1]], dtype=np.float32)

# Step 4: Apply convolution
gx = cv2.filter2D(gray, -1, prewitt_x)
gy = cv2.filter2D(gray, -1, prewitt_y)

# Step 5: Combine both gradients
prewitt_edge = np.sqrt(gx.astype(np.float32)**2 + gy.astype(np.float32)**2)
prewitt_edge = np.uint8(prewitt_edge)

# Step 6: Display results
print("Original Gray Image")
cv2_imshow(gray)

print("Prewitt X Edge")
cv2_imshow(gx)

print("Prewitt Y Edge")
cv2_imshow(gy)

print("Final Prewitt Edge Image (Combined)")
cv2_imshow(prewitt_edge)
