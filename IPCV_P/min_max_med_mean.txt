import cv2
import numpy as np
from google.colab.patches import cv2_imshow

# ---- Read grayscale image ----
img = cv2.imread("/content/sample_data/Lena.jpg", cv2.IMREAD_GRAYSCALE)

# ---- Add Gaussian Noise ----
noise = np.random.normal(0, 20, img.shape)
noisy = np.clip(img + noise, 0, 255).astype(np.uint8)

# ---- Padding for filter operations ----
padded = np.pad(noisy, 1, mode='edge')
h, w = noisy.shape

# ---- Create empty output images ----
mean_img      = np.zeros_like(noisy)
geo_img       = np.zeros_like(noisy)
harmonic_img  = np.zeros_like(noisy)
contra_img    = np.zeros_like(noisy)
median_img    = np.zeros_like(noisy)
min_img       = np.zeros_like(noisy)
max_img       = np.zeros_like(noisy)
midpoint_img  = np.zeros_like(noisy)

Q = 1.5  # Contra-harmonic constant

# ---- Apply all filters (3x3 window) ----
for i in range(1, h+1):
    for j in range(1, w+1):
        
        window = padded[i-1:i+2, j-1:j+2].astype(float)

        mean_img[i-1, j-1]     = np.mean(window)
        geo_img[i-1, j-1]      = np.exp(np.mean(np.log(window + 1)))
        harmonic_img[i-1, j-1] = 9 / np.sum(1/(window + 1e-6))
        contra_img[i-1, j-1]   = np.sum(window**(Q+1)) / np.sum(window**Q)
        median_img[i-1, j-1]   = np.median(window)
        min_img[i-1, j-1]      = np.min(window)
        max_img[i-1, j-1]      = np.max(window)
        midpoint_img[i-1, j-1] = (np.max(window) + np.min(window)) / 2

# ---- Show Results ----
print("Noisy Image:")
cv2_imshow(noisy)

print("Arithmetic Mean Filter:")
cv2_imshow(mean_img)

print("Geometric Mean Filter:")
cv2_imshow(geo_img)

print("Harmonic Mean Filter:")
cv2_imshow(harmonic_img)

print("Contra Harmonic Filter:")
cv2_imshow(contra_img)

print("Median Filter:")
cv2_imshow(median_img)

print("Min Filter:")
cv2_imshow(min_img)

print("Max Filter:")
cv2_imshow(max_img)

print("Midpoint Filter:")
cv2_imshow(midpoint_img)
