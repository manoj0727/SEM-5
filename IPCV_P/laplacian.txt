import cv2
import numpy as np
from google.colab.patches import cv2_imshow
import matplotlib.pyplot as plt

# Step 1: Read image in grayscale
img = cv2.imread('/content/sample_data/filter.tif', cv2.IMREAD_GRAYSCALE)

if img is None:
    raise ValueError("Image not found. Make sure filter.tif exists.")

# Step 2: Define Laplacian Masks

kernel1 = np.array([[0, -1,  0],
                    [-1, 4, -1],
                    [0, -1,  0]], dtype=np.float32)

kernel2 = np.array([[0,  1,  0],
                    [1, -4,  1],
                    [0,  1,  0]], dtype=np.float32)

kernel3 = np.array([[1,  1,  1],
                    [1, -8,  1],
                    [1,  1,  1]], dtype=np.float32)

kernel4 = np.array([[-1, -1, -1],
                    [-1,  8, -1],
                    [-1, -1, -1]], dtype=np.float32)

kernels = [kernel1, kernel2, kernel3, kernel4]
titles = [
    "Laplacian (+4 center)",
    "Laplacian (-4 center)",
    "Laplacian (-8 center)",
    "Laplacian (+8 center)"
]

# Step 3: Apply filters
results = []

for k in kernels:
    filtered = cv2.filter2D(img, -1, k)
    results.append(filtered)

# Step 4: Display results
plt.figure(figsize=(14, 8))
plt.subplot(2, 3, 1)
plt.imshow(img, cmap='gray')
plt.title("Original Image")
plt.axis("off")

for i, (res, title) in enumerate(zip(results, titles)):
    plt.subplot(2, 3, i + 2)
    plt.imshow(res, cmap='gray')
    plt.title(title)
    plt.axis("off")

plt.tight_layout()
plt.show()
