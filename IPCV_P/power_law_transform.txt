import cv2
import numpy as np
from google.colab.patches import cv2_imshow

# Read and convert to grayscale
img = cv2.imread('/content/sample_data/Lena.jpg')
gray = cv2.cvtColor(img, cv2.COLOR_BGR2GRAY)

# Empty image for output
gamma_img = np.zeros_like(gray)

# Constant and Gamma (λ)
c = 1
gamma = 0.6   # change to: 0.2, 4, or 8

# Apply Power Law transform pixel by pixel
for i in range(gray.shape[0]):
    for j in range(gray.shape[1]):
        r = gray[i, j] / 255.0                 # normalize to 0–1
        s = c * (r ** gamma)                   # apply power law formula
        gamma_img[i, j] = np.uint8(s * 255)    # scale back to 0–255

# Show images
cv2_imshow(gray)
cv2_imshow(gamma_img)
