import cv2
import numpy as np
from google.colab.patches import cv2_imshow

# Step 1: Read and convert to grayscale
img = cv2.imread('/content/sample_data/Lena.jpg')
gray = cv2.cvtColor(img, cv2.COLOR_BGR2GRAY)

#########################################
#   WITHOUT BUILT-IN FUNCTION
#########################################

# Step 2: Compute histogram
hist = np.zeros(256)

for i in range(gray.shape[0]):
    for j in range(gray.shape[1]):
        hist[gray[i,j]] += 1

# Step 3: Compute Probability Density Function (PDF)
total_pixels = gray.size
pdf = hist / total_pixels

# Step 4: Compute Cumulative Distribution Function (CDF)
cdf = np.zeros(256)
cdf[0] = pdf[0]
for i in range(1, 256):
    cdf[i] = cdf[i-1] + pdf[i]

# Step 5: Multiply by 255 and round to create mapping
mapping = np.round(cdf * 255).astype(np.uint8)

# Step 6: Create equalized image using mapping
equalized_manual = np.zeros_like(gray)
for i in range(gray.shape[0]):
    for j in range(gray.shape[1]):
        equalized_manual[i, j] = mapping[gray[i, j]]

#########################################
#   WITH BUILT-IN FUNCTION
#########################################

equalized_cv = cv2.equalizeHist(gray)

#########################################
#   DISPLAY RESULTS
#########################################

print("Original Image")
cv2_imshow(gray)

print("Histogram Equalization (Manual)")
cv2_imshow(equalized_manual)

print("Histogram Equalization (Using OpenCV)")
cv2_imshow(equalized_cv)
